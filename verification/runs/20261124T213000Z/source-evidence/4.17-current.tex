\begin{theorem}\label{three:thm:regular-four}
Fix a positive density bound $\alpha$ and ordinary robust parameters
with sufficiently small range. For sufficiently small $\eps>0$
and large even $n$, let $R$ be a simple $k$-regular graph of order
$n$ with no $k$-overfull set, where $k\ge\alpha n$ and
\[
 V(R)=U_1\cup\cdots\cup U_q\cup Z,\qquad q\in\{3,4\},
 \qquad |U_i|\ge\alpha n.
\]
If the $U_i$ contain cores of error $a$ and
$a+|Z|+1\le\eps n$, then $R$ has a $1$-factorization.
\end{theorem}
