\begin{theorem}\label{three:thm:main}
Let $\alpha>0$. For every sufficiently small $\tau>0$, depending
on $\alpha$, and every $0<\nu\le\tau$, there exist $\eps>0$ and
$n_0\in\mathbb N$ for which the following holds. Let $G$ be a
graph of order $n\ge n_0$, with $\Delta(G)\le k$ and $k\ge\alpha n$.
Suppose
\[
 V(G)=U_1\cup\cdots\cup U_q\cup Z,\qquad q\in\{1,2,3\},
 \qquad |U_i|\ge\alpha n,\qquad a+|Z|+1\le\eps n,
\]
and each $G[U_i]$ contains a spanning robust $(\nu,\tau)$-expander
of minimum degree at least $k-a$. If $G$ has no $k$-overfull set,
then $G$ is $k$-edge-colorable.
\end{theorem}
