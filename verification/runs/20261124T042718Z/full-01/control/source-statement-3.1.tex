\begin{lemma}\label{single:lem:coloring}
Fix ordinary robust parameters $0<\nu\le\tau$ with sufficiently
small range. There are $\eps>0$ and $n_0$ such that the following
holds. Let $G$ be a simple graph of order $n\ge n_0$ and maximum
degree $\Delta\ge n/4$, with no $\Delta$-overfull set. Suppose
\[
 V(G)=U\cup Z,\qquad |U|\ge n/4,\qquad
 a+|Z|+1\le\eps n,
\]
and $G[U]$ contains a spanning robust $(\nu,\tau)$-expander
of minimum degree at least $\Delta-a$. Then $G$ is $\Delta$-edge-colorable.
\end{lemma}
